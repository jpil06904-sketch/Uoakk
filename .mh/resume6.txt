def show(label, pat, before=100, after=1500, maxn=3):
    ms = list(re.finditer(pat, s))
    out.append('### %s %d' % (label, len(ms)))
    for m in ms[:maxn]:
        st = max(0, m.start() - before)
        out.append('  @%d :: %s' % (m.start(), s[st:m.start() + after].replace('\n', ' \u23ce ')))
